function doGet(e) { return ContentService.createTextOutput("School Duty API Running..."); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var data = JSON.parse(e.postData.contents); var action = data.action; var result = {}; if (action === 'getInitialData') result = getInitialData(); else if (action === 'getStudentsByLocation') result = getStudentsByLocation(data.location); else if (action === 'submitReport') result = submitReport(data); else if (action === 'getReportsForStudent') result = getReportsForStudent(data.location); else if (action === 'getPendingReports') result = getPendingReports(); else if (action === 'gradeReport') result = gradeReport(data); else if (action === 'getAdminStats') result = getAdminStats(); else if (action === 'manageLocation') result = manageLocation(data); else if (action === 'manageTeacher') result = manageTeacher(data); else if (action === 'randomDuty') result = randomDuty(data); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function getSS() { return SpreadsheetApp.getActiveSpreadsheet(); } // 1. ดึงข้อมูลเริ่มต้น (สถานที่ + รายชื่อครู) function getInitialData() { var ss = getSS(); var locSheet = ss.getSheetByName('Locations'); var teacherSheet = ss.getSheetByName('Teachers'); var locations = []; if (locSheet) { var values = locSheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0]) locations.push(values[i][0].toString().trim()); } } var teachers = []; if (teacherSheet) { var values = teacherSheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0]) teachers.push(values[i][0].toString().trim()); } } return { success: true, locations: locations, teachers: teachers }; } // 2. ดึงรายชื่อนักเรียนตามจุด (อ่านคอลัมน์ B=ชื่อ, C=สถานที่ + ป้องกันเรื่องเว้นวรรค) function getStudentsByLocation(location) { var ss = getSS(); var sheet = ss.getSheetByName('Students'); var students = []; if (sheet) { var values = sheet.getDataRange().getValues(); // ตัดเว้นวรรคทั้งหมดออกเพื่อเปรียบเทียบ var targetLocation = location ? location.toString().replace(/\s+/g, '').toLowerCase() : ''; for (var i = 1; i < values.length; i++) { var studentName = values[i][1] ? values[i][1].toString().trim() : ''; // คอลัมน์ B (ชื่อ-นามสกุล) var studentLoc = values[i][2] ? values[i][2].toString().replace(/\s+/g, '').toLowerCase() : ''; // คอลัมน์ C (สถานที่) if (studentLoc === targetLocation && studentName !== '') { students.push(studentName); } } } return { success: true, students: students }; } // 3. ส่งรายงานการทำเวร function submitReport(data) { var ss = getSS(); var sheet = ss.getSheetByName('Reports') || ss.insertSheet('Reports'); if (sheet.getLastRow() === 0) { sheet.appendRow(['ID', 'Timestamp', 'StudentName', 'Location', 'Status', 'AbsentDetails', 'PhotoUrl', 'InspectionStatus', 'Score', 'Feedback', 'TeacherName']); } var id = 'REP_' + new Date().getTime(); var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm"); sheet.appendRow([ id, timestamp, data.studentName, data.location, data.status, data.absentDetails || '-', data.photoUrl || '-', 'รอการตรวจ', '-', '-', '-' ]); return { success: true }; } // 4. นักเรียนดูผลการตรวจและคำแนะนำจากครู function getReportsForStudent(location) { var ss = getSS(); var sheet = ss.getSheetByName('Reports'); var list = []; if (sheet) { var values = sheet.getDataRange().getValues(); var targetLoc = location ? location.toString().replace(/\s+/g, '').toLowerCase() : ''; for (var i = values.length - 1; i >= 1; i--) { var sheetLoc = values[i][3] ? values[i][3].toString().replace(/\s+/g, '').toLowerCase() : ''; if (sheetLoc === targetLoc) { list.push({ id: values[i][0], timestamp: values[i][1], studentName: values[i][2], status: values[i][4], absentDetails: values[i][5], inspectionStatus: values[i][7], score: values[i][8], feedback: values[i][9], teacherName: values[i][10] }); } } } return { success: true, reports: list }; } // 5. ครูดึงรายการที่รอการตรวจ function getPendingReports() { var ss = getSS(); var sheet = ss.getSheetByName('Reports'); var list = []; if (sheet) { var values = sheet.getDataRange().getValues(); for (var i = values.length - 1; i >= 1; i--) { list.push({ id: values[i][0], timestamp: values[i][1], studentName: values[i][2], location: values[i][3], status: values[i][4], absentDetails: values[i][5], photoUrl: values[i][6], inspectionStatus: values[i][7], score: values[i][8], feedback: values[i][9], teacherName: values[i][10], rowIndex: i + 1 }); } } return { success: true, reports: list }; } // 6. ครูบันทึกการตรวจ + คะแนน + คำแนะนำ function gradeReport(data) { var ss = getSS(); var sheet = ss.getSheetByName('Reports'); if (!sheet) return { success: false }; var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0] === data.id) { sheet.getRange(i + 1, 8).setValue('ตรวจแล้ว'); sheet.getRange(i + 1, 9).setValue(data.score); sheet.getRange(i + 1, 10).setValue(data.feedback); sheet.getRange(i + 1, 11).setValue(data.teacherName); return { success: true }; } } return { success: false }; } // 7. สถิติแอดมิน (นับจำนวนรายงานแยกตามสถานที่) function getAdminStats() { var ss = getSS(); var sheet = ss.getSheetByName('Reports'); var counts = {}; if (sheet) { var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { var loc = values[i][3] ? values[i][3].toString().trim() : ''; if (loc) counts[loc] = (counts[loc] || 0) + 1; } } return { success: true, stats: counts }; } // 8. เพิ่ม / ลด สถานที่ function manageLocation(data) { var ss = getSS(); var sheet = ss.getSheetByName('Locations') || ss.insertSheet('Locations'); if (data.type === 'add') { sheet.appendRow([data.name.toString().trim()]); } else if (data.type === 'delete') { var values = sheet.getDataRange().getValues(); for (var i = 0; i < values.length; i++) { if (values[i][0].toString().trim() === data.name.toString().trim()) { sheet.deleteRow(i + 1); break; } } } return getInitialData(); } // 9. เพิ่ม / ลด รายชื่อครู function manageTeacher(data) { var ss = getSS(); var sheet = ss.getSheetByName('Teachers') || ss.insertSheet('Teachers'); if (data.type === 'add') { sheet.appendRow([data.name.toString().trim()]); } else if (data.type === 'delete') { var values = sheet.getDataRange().getValues(); for (var i = 0; i < values.length; i++) { if (values[i][0].toString().trim() === data.name.toString().trim()) { sheet.deleteRow(i + 1); break; } } } return getInitialData(); } // 10. สุ่มรายชื่อเวรประจำจุด function randomDuty(data) { var ss = getSS(); var sheet = ss.getSheetByName('Students'); var allStudents = []; if (sheet) { var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][1]) allStudents.push(values[i][1].toString().trim()); } } var shuffled = allStudents.sort(() => 0.5 - Math.random()); var result = {}; data.locations.forEach(function(loc) { result[loc] = shuffled.splice(0, 2); }); return { success: true, result: result }; }
