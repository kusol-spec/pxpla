// รองรับ CORS Preflight สำหรับ Browser function doOptions(e) { return ContentService.createTextOutput("") .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { if (e && e.parameter && e.parameter.action) { return handleActions(e.parameter.action, e.parameter); } return ContentService.createTextOutput("School Duty API is Running..."); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var data = {}; if (e && e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } var result = handleActions(data.action, data); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function handleActions(action, data) { if (action === 'getInitialData') return getInitialData(); if (action === 'submitReport') return submitReport(data); if (action === 'getStudentReports') return getStudentReports(); if (action === 'teacherLogin') return teacherLogin(data); if (action === 'getPendingReports') return getPendingReports(); if (action === 'gradeReport') return gradeReport(data); if (action === 'getAdminOverview') return getAdminOverview(); if (action === 'addLocation') return addLocation(data); if (action === 'deleteLocation') return deleteLocation(data); if (action === 'addTeacher') return addTeacher(data); if (action === 'deleteTeacher') return deleteTeacher(data); return { success: false, message: "Unknown action" }; } function getSpreadsheet() { return SpreadsheetApp.getActiveSpreadsheet(); } function getInitialData() { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Locations'); var locations = []; if (sheet) { var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0]) locations.push(values[i][0].toString()); } } return { success: true, locations: locations }; } function submitReport(data) { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Reports'); if (!sheet) { sheet = ss.insertSheet('Reports'); sheet.appendRow(['ID', 'Timestamp', 'StudentName', 'Location', 'Status', 'AbsentDetails', 'PhotoUrl', 'InspectionStatus', 'Score', 'Feedback', 'TeacherName']); } var id = 'REP_' + new Date().getTime(); var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm"); var photoUrl = ""; if (data.photoData && data.photoData.indexOf('base64,') !== -1) { photoUrl = saveImageToDrive(data.photoData, id); } sheet.appendRow([ id, timestamp, data.studentName, data.location, data.status, data.absentDetails || '-', photoUrl, 'รอการตรวจ', '-', '-', '-' ]); return { success: true, id: id }; } function saveImageToDrive(base64Data, filename) { try { var folderName = "School_Duty_Photos"; var folders = DriveApp.getFoldersByName(folderName); var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var splitData = base64Data.split('base64,'); var contentType = splitData[0].split(':')[1].split(';')[0]; var bytes = Utilities.base64Decode(splitData[1]); var blob = Utilities.newBlob(bytes, contentType, filename + ".jpg"); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return "https://lh3.googleusercontent.com/d/" + file.getId(); } catch (e) { return ""; } } function getStudentReports() { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Reports'); if (!sheet) return { success: true, reports: [] }; var data = sheet.getDataRange().getValues(); var reports = []; for (var i = data.length - 1; i >= 1; i--) { reports.push({ id: data[i][0], timestamp: data[i][1], studentName: data[i][2], location: data[i][3], status: data[i][4], absentDetails: data[i][5], photoUrl: data[i][6], inspectionStatus: data[i][7], score: data[i][8], feedback: data[i][9] }); } return { success: true, reports: reports }; } function teacherLogin(data) { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Teachers'); if (!sheet) return { success: false }; var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][1] == data.username && values[i][2] == data.password) { return { success: true, teacherName: values[i][0] }; } } return { success: false }; } function getPendingReports() { var res = getStudentReports(); var pending = res.reports.filter(function(r) { return r.inspectionStatus === 'รอการตรวจ'; }); return { success: true, reports: pending }; } function gradeReport(data) { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Reports'); if (!sheet) return { success: false }; var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0] === data.id) { sheet.getRange(i + 1, 8).setValue('ตรวจแล้ว'); sheet.getRange(i + 1, 9).setValue(data.score); sheet.getRange(i + 1, 10).setValue(data.feedback); sheet.getRange(i + 1, 11).setValue(data.teacherName); return { success: true }; } } return { success: false }; } function getAdminOverview() { var ss = getSpreadsheet(); var reportSheet = ss.getSheetByName('Reports'); var reports = reportSheet ? reportSheet.getDataRange().getValues() : []; var total = reports.length - 1; var inspected = 0, pending = 0; for (var i = 1; i < reports.length; i++) { if (reports[i][7] === 'ตรวจแล้ว') inspected++; else if (reports[i][7] === 'รอการตรวจ') pending++; } var locRes = getInitialData(); var teacherSheet = ss.getSheetByName('Teachers'); var teachersData = teacherSheet ? teacherSheet.getDataRange().getValues() : []; var teachers = []; for (var j = 1; j < teachersData.length; j++) { teachers.push({ name: teachersData[j][0], username: teachersData[j][1] }); } return { success: true, stats: { total: total > 0 ? total : 0, inspected: inspected, pending: pending }, locations: locRes.locations, teachers: teachers }; } function addLocation(data) { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Locations'); if (!sheet) { sheet = ss.insertSheet('Locations'); sheet.appendRow(['LocationName']); } sheet.appendRow([data.location]); return { success: true }; } function deleteLocation(data) { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Locations'); if (!sheet) return { success: false }; var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0] === data.location) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; } function addTeacher(data) { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Teachers'); if (!sheet) { sheet = ss.insertSheet('Teachers'); sheet.appendRow(['Name', 'Username', 'Password']); } sheet.appendRow([data.name, data.username, data.password]); return { success: true }; } function deleteTeacher(data) { var ss = getSpreadsheet(); var sheet = ss.getSheetByName('Teachers'); if (!sheet) return { success: false }; var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][1] === data.username) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; }
