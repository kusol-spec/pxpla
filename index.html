// ===== ระบบตรวจความสะอาด: Backend (Google Apps Script) ===== // วางใน Google Sheets > ส่วนขยาย > Apps Script แล้วรันฟังก์ชัน setup() 1 ครั้งก่อน Deploy const TZ = 'Asia/Bangkok'; const FOLDER = 'รูปงานนักเรียน'; const DEFAULT_ADMIN_PW = '8888'; const SUB = ['id','date','sender','class','area','image','status','score','comment','checker','checkedAt']; const TCH = ['username','name','hash','approved']; function fmt(s, n) { s.getRange(1, 1, s.getMaxRows(), n).setNumberFormat('@'); s.setFrozenRows(1); } function sheet(n, h) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let s = ss.getSheetByName(n); if (!s) { s = ss.insertSheet(n); s.appendRow(h); fmt(s, h.length); } return s; } function add(s, vals) { // เพิ่มแถวแบบเก็บเป็นข้อความ กัน Sheets แปลงวันที่/ตัวเลขเอง s.insertRowAfter(s.getLastRow()); s.getRange(s.getLastRow() + 1, 1, 1, vals.length).setNumberFormat('@').setValues([vals]); } function setup() { // รันครั้งเดียวเพื่อสร้างชีต โฟลเดอร์ และอนุญาตสิทธิ์ fmt(sheet('Submissions', SUB), SUB.length); fmt(sheet('Teachers', TCH), TCH.length); folder(); } function folder() { const it = DriveApp.getFoldersByName(FOLDER); return it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER); } function rows(s) { const v = s.getDataRange().getValues(), h = v.shift(); return v.map((r, i) => { const o = { _row: i + 2 }; h.forEach((k, j) => o[k] = r[j] instanceof Date ? Utilities.formatDate(r[j], TZ, 'yyyy-MM-dd HH:mm') : r[j]); return o; }); } function sha(s) { return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s, Utilities.Charset.UTF_8) .map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join(''); } function adminHash() { return sha(PropertiesService.getScriptProperties().getProperty('ADMIN_PASSWORD') || DEFAULT_ADMIN_PW); } function auth(d, need) { if (d.role === 'admin') { if (d.hash !== adminHash()) throw new Error('รหัสผ่านแอดมินไม่ถูกต้อง'); return { name: 'แอดมิน', admin: true }; } if (need === 'admin') throw new Error('ต้องเป็นแอดมินเท่านั้น'); const t = rows(sheet('Teachers', TCH)).find(r => r.username === d.user && r.hash === d.hash); if (!t) throw new Error('ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง'); if (t.approved !== 'yes') throw new Error('บัญชียังไม่ได้รับการอนุมัติจากแอดมิน'); return { name: t.name }; } function clean(v, max) { return String(v == null ? '' : v).trim().slice(0, max || 200); } function saveImg(b64, id) { const m = String(b64 || '').match(/^data:(image\/[\w+.-]+);base64,(.+)$/); if (!m) throw new Error('ไฟล์รูปภาพไม่ถูกต้อง'); const f = folder().createFile(Utilities.newBlob(Utilities.base64Decode(m[2]), m[1], id + '.jpg')); try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} return 'https://drive.google.com/thumbnail?id=' + f.getId() + '&sz=w1000'; } function delImg(url) { const m = String(url || '').match(/id=([^&]+)/); if (m) try { DriveApp.getFileById(m[1]).setTrashed(true); } catch (e) {} } function handle(d) { const S = sheet('Submissions', SUB), T = sheet('Teachers', TCH); switch (d.action) { case 'submit': { const sender = clean(d.sender, 100), area = clean(d.area, 150); if (!sender || !area) throw new Error('กรอกชื่อและสถานที่ให้ครบ'); const id = 'S' + Utilities.getUuid().replace(/-/g, '').slice(0, 8); add(S, [id, clean(d.date, 10), sender, clean(d.class, 30), area, saveImg(d.image, id), 'รอตรวจ', '', '', '', '']); return { id }; } case 'status': { // ต้องพิมพ์ชื่อให้ตรงเท่านั้น กันเห็นข้อมูลคนอื่น const q = clean(d.sender, 100).toLowerCase(); if (!q) return []; return rows(S).filter(r => String(r.sender).trim().toLowerCase() === q) .map(r => ({ date: r.date, sender: r.sender, area: r.area, status: r.status, score: r.score, comment: r.comment, checker: r.checker })); } case 'register': { const user = clean(d.user, 40), name = clean(d.name, 80); if (!user || !name || !d.hash) throw new Error('กรอกข้อมูลให้ครบ'); if (rows(T).some(r => r.username === user)) throw new Error('ชื่อผู้ใช้นี้ถูกใช้แล้ว'); add(T, [user, name, d.hash, 'no']); return {}; } case 'login': return auth(d); case 'list': auth(d); return rows(S).reverse(); case 'grade': { const me = auth(d), r = rows(S).find(x => x.id === d.id), sc = Number(d.score); if (!r) throw new Error('ไม่พบรายการ'); if (isNaN(sc) || sc < 0 || sc > 10) throw new Error('คะแนนต้องอยู่ระหว่าง 0-10'); S.getRange(r._row, 7, 1, 5).setValues([['ตรวจแล้ว', sc, clean(d.comment, 500), me.name, Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm')]]); return {}; } case 'teachers': auth(d, 'admin'); return rows(T).map(r => ({ username: r.username, name: r.name, approved: r.approved })); case 'addTeacher': { auth(d, 'admin'); const user = clean(d.user, 40), name = clean(d.name, 80); if (!user || !name || !d.newHash) throw new Error('กรอกข้อมูลให้ครบ'); if (rows(T).some(r => r.username === user)) throw new Error('ชื่อผู้ใช้นี้ถูกใช้แล้ว'); add(T, [user, name, d.newHash, 'yes']); return {}; } case 'approve': { auth(d, 'admin'); const r = rows(T).find(x => x.username === d.user); if (r) T.getRange(r._row, 4).setValue('yes'); return {}; } case 'resetPw': { auth(d, 'admin'); const r = rows(T).find(x => x.username === d.user); if (!r) throw new Error('ไม่พบครู'); T.getRange(r._row, 3).setValue(d.newHash); return {}; } case 'delTeacher': { auth(d, 'admin'); const r = rows(T).find(x => x.username === d.user); if (r) T.deleteRow(r._row); return {}; } case 'delSub': { auth(d, 'admin'); const r = rows(S).find(x => x.id === d.id); if (r) { delImg(r.image); S.deleteRow(r._row); } return {}; } case 'setAdminPw': { auth(d, 'admin'); if (!d.newPw || d.newPw.length < 4) throw new Error('รหัสผ่านต้องยาวอย่างน้อย 4 ตัวอักษร'); PropertiesService.getScriptProperties().setProperty('ADMIN_PASSWORD', d.newPw); return {}; } } throw new Error('ไม่รู้จักคำสั่ง'); } function doPost(e) { const lock = LockService.getScriptLock(); let out; try { lock.waitLock(25000); out = { result: 'success', data: handle(JSON.parse(e.postData.contents)) }; } catch (err) { out = { result: 'error', message: String(err.message || err) }; } finally { try { lock.releaseLock(); } catch (x) {} } return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); }
